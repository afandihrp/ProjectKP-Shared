// src/App.jsx

import { useState, useEffect } from 'react';
import reactLogo from './assets/react.svg';
import viteLogo from '/vite.svg';
import './App.css';
import LandingPage from './LandingPage';

function App() {
  // State to hold the message from the backend
  const [message, setMessage] = useState('');
  
  // State to track if the data is still loading
  const [isLoading, setIsLoading] = useState(true);
  
  // State to hold any potential errors
  const [error, setError] = useState(null);

  useEffect(() => {
    // Fetch data from the backend API
    fetch('/api/') // This matches the endpoint in server.js
      .then(response => {
        // If the response is not successful, throw an error
        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }
        // Otherwise, parse the JSON
        return response.json();
      })
      .then(data => {
        // Set the message from the data and turn off loading
        setMessage(data.message);
        setIsLoading(false);
      })
      .catch(error => {
        // If an error occurs, save the error and turn off loading
        console.error("There was an error fetching the data:", error);
        setError(error.message);
        setIsLoading(false);
      });
  }, []); // The empty array ensures this effect runs only once after the component mounts

  return (
    <>
      <div>
        <a href="https://vitejs.dev" target="_blank" rel="noopener noreferrer">
          <img src={viteLogo} className="logo" alt="Vite logo" />
        </a>
        <a href="https://react.dev" target="_blank" rel="noopener noreferrer">
          <img src={reactLogo} className="logo react" alt="React logo" />
        </a>
      </div>
      <h1>Vite + React + Express</h1>
      <div className="card">
        {/* Conditional Rendering Section */}
        {isLoading ? (
          <p>Loading message from backend...</p>
        ) : error ? (
          <p style={{ color: 'red' }}>Error: {error}</p>
        ) : (
          <p><strong>Backend says:</strong> {message}</p>
        )}
      </div>
      <p className="read-the-docs">
        Your full-stack app is running!
      </p>
    </>
  );
}

export default App;
